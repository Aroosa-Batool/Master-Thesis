# Python dependencies for the `robot` package.
# Target interpreter: Python 3.13. (bleak 0.22 has no 3.14 wheel, so 3.14 fails
# to resolve; create the venv with `python3.13 -m venv .venv`.)
# Install into a per-machine venv: `pip install -r requirements.txt`.

# Array math used directly across the whole stack (embeddings, RMS, cosine).
# Also pulled in transitively by opencv/resemblyzer, but every module imports
# it directly, so declare it explicitly.
numpy>=1.24

# Webcam capture + face detection (presence sensing).
opencv-python>=4.8,<5

# BLE client for talking to the Bangle.js watch.
# bleak abstracts over WinRT (Windows) and CoreBluetooth (macOS) so the
# same code runs on both laptops.
bleak>=0.22,<1

# Ohbot SDK (used by the reminder-delivery apps under robot/apps/).
ohbot>=2.0,<3

# --- Voice modality (robot/core/robot_io.py + robot/apps/enroll_voice.py) ---
# Microphone capture. Needs the PortAudio system library (bundled in the
# macOS/Windows wheels; on Linux: `apt install libportaudio2`).
sounddevice>=0.4,<1
# Pretrained GE2E speaker embeddings ("d-vector"). Pulls in torch + librosa
# + webrtcvad transitively; model weights ship inside the package, so there
# is no separate download.
resemblyzer>=0.1.3,<0.2
# resemblyzer -> webrtcvad still imports the legacy `pkg_resources` module at
# import time, which setuptools removed in 81.0. Pin below that so
# `import webrtcvad` works (fresh Python 3.13 venvs also ship no setuptools).
setuptools<81
# Neural voice-activity detection (robot/perception/speech_vad.py): decides
# whether a sound is a HUMAN VOICE at all, which the RMS energy gate cannot.
# Without it a grinder / fan / vacuum is loud, is not the owner, and so gets
# filed as a bystander - resemblyzer's own webrtcvad(mode 3) does NOT catch this
# (measured: it keeps 99.8% of pure broadband noise as "speech"). Weights ship
# inside the package, so there is no download and no network use at run time.
silero-vad>=5,<7

# --- Voice reminders (robot/apps/add_reminder.py: "ask the robot to add a reminder") ---
# Speech-to-text for the spoken "add a reminder" command. Uses torch (already
# a resemblyzer dep); downloads its model (~74 MB for base.en) into
# ~/.cache/whisper on first run.
openai-whisper>=20231117
# Pulls the date/time out of the spoken command's natural language
# ("July 20th at 3 PM", "tomorrow at 2:30 PM").
dateparser>=1.2,<2

# --- Reminder sensitivity classifier (robot/core/sensitivity.py) ---
# Local sentence embeddings to label a reminder sensitive vs non-sensitive,
# which gates whether it needs presence + watch consent before being spoken.
# Reuses the resemblyzer torch install; downloads a small (~80 MB) MiniLM model
# to the HuggingFace cache on first run, then runs offline on CPU. Optional:
# sentence-transformers falls back to a keyword heuristic if it is not installed.
sentence-transformers>=2.2,<6

# --- Development / tests (tests/) ---
# The hardware-free test suite (run with `python -m pytest tests/`). Uses
# mocks/fakes throughout, so no microphone, camera, watch, Ohbot, model
# download, or network is needed to run it.
pytest>=8
